#   pip install -r requirements.txt -r requirements-rag.txt

# Lưu ý cho máy Windows: src/rag/parse.py tự đặt TORCH_COMPILE_DISABLE=1, vì

# Visual Studio Build Tools thì không có, và nó sẽ làm hỏng cả lần parse.
